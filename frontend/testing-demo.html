Cypress: prueba E2E (slide 8) 

describe('Crear pedido', () => {
  it('registra el pedido y confirma el orderId', () => {
    cy.visit('/');
    cy.get('[data-cy=business-card]').first().click();
    cy.get('[data-cy=add-to-cart]').click();
    cy.get('[data-cy=confirm-order]').click();
    cy.url().should('include', '/recibo');
    cy.get('[data-cy=order-id]').should('be.visible');
  });
});

cy.request(): probar la API (slide 11) 

it('crea un pedido y devuelve su identificador', () => {
  cy.request({
    method: 'POST',
    url: '/api/orders',
    body: {
      businessId: 'biz_42',
      products: [{ name: 'Cafe', price: 8000, qty: 2 }],
      resident: {
        fullName: 'Ana', phone: '3001234567',
        tower: '1', apartment: '101',
        deliveryAddress: 'Mz A Casa 1'
      }
    },
    failOnStatusCode: false
  }).then((res) => {
    expect(res.status).to.eq(201);
    expect(res.body.data).to.have.property('orderId');
    expect(res.body.data.estado).to.eq('Recibido');
  });
});
